/* Chat component variables
   Tier 2 exception — bubble surfaces, avatar, message box border
   have no direct SHARED SEMANTIC equivalent.
   All vars reference semantic tokens. Raw hex only in dark override.
   Figma collection: COMPONENTS · Path prefix: Chat/ */
:root {
  /* Container */
  --chat-bg:                       var(--color-surface-default);
  --chat-border:                   var(--color-border-default);

  /* Received bubble (alignment-start) */
  --chat-bubble-received-bg:       var(--color-theme-10);         /* → #f2f2f2 */
  --chat-bubble-received-text:     var(--color-text-default);     /* → #3f3e3d */

  /* Sent bubble (alignment-end) */
  --chat-bubble-sent-bg:           var(--color-brand-default);    /* → #fcd515 */
  --chat-bubble-sent-text:         var(--color-theme-80);         /* → #3f3e3d */

  /* Avatar */
  --chat-avatar-bg:                var(--color-theme-30);         /* → #cfcece */
  --chat-avatar-text:              var(--color-text-default);

  /* Metadata */
  --chat-username-text:            var(--color-text-subtle);      /* → #6f6e6d */
  --chat-timestamp-text:           var(--color-text-subtle);

  /* Day header */
  --chat-dayheader-text:           var(--color-text-subtle);
  --chat-dayheader-line:           var(--color-border-default);

  /* Message box */
  --chat-messagebox-bg:            var(--color-surface-default);
  --chat-messagebox-border:        var(--color-theme-30);         /* → #cfcece */
  --chat-input-border:             var(--color-theme-30);
  --chat-input-border-focus:       var(--color-border-focus);     /* → #1a88b7 */
  --chat-input-text:               var(--color-text-default);
  --chat-input-placeholder:        var(--color-text-subtle);
  --chat-focus-ring:               var(--color-border-focus);     /* → #1a88b7 */

  /* Send button */
  --chat-send-bg:                  var(--color-brand-default);    /* → #fcd515 */
  --chat-send-bg-hover:            var(--color-brand-hover);      /* → #fbe67e */
  --chat-send-icon:                var(--color-theme-80);         /* → #3f3e3d */
}

}
