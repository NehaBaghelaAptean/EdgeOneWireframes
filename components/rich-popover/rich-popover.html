/* Rich Popover component variables
   Tier 1 component — nearly all roles cascade from SHARED SEMANTIC.
   Tier 2 exception: --rich-popover-shadow (no semantic elevation token).
   Figma collection: COMPONENTS · Path prefix: Rich Popover/ */
:root {
  /* Panel */
  --rich-popover-surface:           var(--color-surface-default);    /* → #ffffff */
  --rich-popover-border:            var(--color-border-default);     /* → #e7e6e6 */
  --rich-popover-shadow:            0 4px 16px rgba(0,0,0,0.12), 0 1px 4px rgba(0,0,0,0.06);

  /* Arrow */
  --rich-popover-arrow-bg:          var(--color-surface-default);
  --rich-popover-arrow-border:      var(--color-border-default);

  /* Header */
  --rich-popover-title-color:       var(--color-text-default);       /* Text/Default → #3f3e3d */
  --rich-popover-header-divider:    var(--color-border-default);

  /* Body */
  --rich-popover-body-color:        var(--color-text-default);       /* Text/Default → #3f3e3d */
  --rich-popover-body-color-subtle: var(--color-text-subtle);        /* Text/Subtle  → #6f6e6d */

  /* Footer */
  --rich-popover-footer-divider:    var(--color-border-default);

  /* Close button */
  --rich-popover-close-icon:        var(--color-icon-default);       /* → #878686 */
  --rich-popover-close-icon-hover:  var(--color-icon-hover);         /* → #3f3e3d */
  --rich-popover-close-bg-hover:    var(--color-surface-hover);      /* → #e7e6e6 */
  --rich-popover-close-focus-ring:  #1a88b7;                         /* Border/Focus (SHARED SEMANTIC) */
}

/* Dark mode — TDK. † = explicit override (does not cascade) */
html[data-theme="dark"] {
  --color-surface-default:          #31313b;
  --color-surface-hover:            #3c3c44;
  --color-text-default:             #d4d4d5;
  --color-text-subtle:              #a9a9ac;
  --color-border-default:           #3c3c44;
  --color-icon-default:             #bcbcbe;
  --color-icon-hover:               #d4d4d5;

  --rich-popover-shadow:            0 4px 16px rgba(0,0,0,0.32), 0 1px 4px rgba(0,0,0,0.20); /* † */
  --rich-popover-close-focus-ring:  #7c79ff; /* † Border/Focus TDK */
}
